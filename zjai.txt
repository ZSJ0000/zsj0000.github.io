# -*- coding: utf-8 -*-

import importlib.util
import json
import os
import platform
import sys
from datetime import datetime, timedelta, timezone

import httpx
from openai import OpenAI

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
CONFIG_PATH = os.path.join(BASE_DIR, "config.json")
SKILL_DIR = os.path.join(BASE_DIR, "skills")

LOGO = """
################################
#                              #
#          ZJAI Agent          #
#                              #
################################
"""


def now_beijing():
    return datetime.now(
        timezone(timedelta(hours=8))
    ).strftime("%Y-%m-%d %H:%M:%S")


def stop(message):
    print(message)
    sys.exit(1)


def normalize_api(item, index):
    if not isinstance(item, dict):
        stop("API {} 配置必须是对象。".format(index + 1))
    item = item.copy()
    item["name"] = str(item.get("name", index + 1))
    item["mode"] = str(item.get("mode", "OPAI")).upper()
    if item["mode"] in ("OPENAI", "OPENAI_COMPATIBLE"):
        item["mode"] = "OPAI"
    if item["mode"] not in ("OPAI", "POST"):
        stop("API {} 的 mode 只能是 OPAI 或 POST。".format(item["name"]))
    for key in ("base_url", "model_name"):
        if not item.get(key):
            stop("API {} 的 {} 不能为空。".format(item["name"], key))
    item["api_key"] = str(item.get("api_key") or "").strip()
    item["max_steps"] = max(1, int(item.get("max_steps", 10)))
    item["temperature"] = max(0.0, float(item.get("temperature", 0.1)))
    item["verify_ssl"] = bool(item.get("verify_ssl", True))
    return item


def load_config():
    if not os.path.exists(CONFIG_PATH):
        stop("找不到 config.json，请先创建配置文件。")
    try:
        with open(CONFIG_PATH, "r", encoding="utf-8") as f:
            data = json.load(f)
    except Exception as e:
        stop("读取 config.json 失败：{}".format(e))
    if "apis" not in data:
        data = {
            "apis": [{
                "name": "1",
                "api_key": data.get("api_key", ""),
                "base_url": data.get("base_url", ""),
                "model_name": data.get("model_name", ""),
                "mode": data.get("mode", "OPAI"),
                "max_steps": data.get("max_steps", 10),
                "temperature": data.get("temperature", 0.1),
                "verify_ssl": data.get("verify_ssl", True)
            }],
            "active_api": "1",
            "快速启动": data.get("快速启动", "False"),
            "agent_name": data.get("agent_name", "ZJAI Agent")
        }
    apis = [normalize_api(x, i) for i, x in enumerate(data.get("apis", []))]
    if not apis:
        stop("config.json 的 apis 不能为空。")
    names = [x["name"] for x in apis]
    active = str(data.get("active_api", names[0]))
    if active not in names:
        active = names[0]
    return {
        "apis": apis,
        "active_api": active,
        "快速启动": str(data.get("快速启动", "False")),
        "agent_name": str(data.get("agent_name", "ZJAI Agent"))
    }


def load_skills():
    tools = []
    functions = {}
    if not os.path.isdir(SKILL_DIR):
        return tools, functions
    for name in sorted(os.listdir(SKILL_DIR)):
        if not (name.startswith("skill_") and name.endswith(".py")):
            continue
        path = os.path.join(SKILL_DIR, name)
        try:
            spec = importlib.util.spec_from_file_location(name[:-3], path)
            module = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(module)
            if hasattr(module, "TOOL_DEF"):
                tools.append(module.TOOL_DEF)
            if hasattr(module, "TOOL_DEFS"):
                tools.extend(module.TOOL_DEFS)
            if hasattr(module, "TOOL_FUNC"):
                functions.update(module.TOOL_FUNC)
            print("已加载 Skill：{}".format(name))
        except Exception as e:
            print("加载 Skill 失败：{}，错误：{}".format(name, e))
    return tools, functions


CONFIG = load_config()
ALL_TOOLS, ALL_TOOL_MAP = load_skills()
API_MAP = {x["name"]: x for x in CONFIG["apis"]}
CURRENT_API_NAME = CONFIG["active_api"]
AGENT_NAME = CONFIG["agent_name"]

if platform.system() == "Windows":
    OS_TEXT = "Microsoft Windows"
    CMD_TEXT = "Windows CMD"
    CMD_RULE = "Windows 命令使用 dir、type；禁止使用 ls、cat、pwd、rm。"
else:
    OS_TEXT = platform.system()
    CMD_TEXT = "当前系统命令行"
    CMD_RULE = "请根据当前系统使用正确命令。"

SYSTEM_PROMPT = (
    "你是插件化 AI Agent。"
    "名称：{}。操作系统：{}。命令环境：{}。{}"
    "必须使用简体中文回答。"
    "不得编造工具执行结果。只有看到工具返回后，才能声称操作完成。"
    "如果需要工具，必须调用工具。"
).format(AGENT_NAME, OS_TEXT, CMD_TEXT, CMD_RULE)

CHAT_MESSAGES = [{"role": "system", "content": SYSTEM_PROMPT}]


def current_api():
    return API_MAP[CURRENT_API_NAME]


def update_time():
    text = CHAT_MESSAGES[0]["content"]
    marker = "当前北京时间："
    line = marker + now_beijing()
    start = text.find(marker)
    if start >= 0:
        end = text.find("\n", start)
        if end < 0:
            end = len(text)
        text = text[:start] + line + text[end:]
    else:
        text += "\n" + line
    CHAT_MESSAGES[0]["content"] = text


def execute_tool(name, arguments):
    fn = ALL_TOOL_MAP.get(name)
    if fn is None:
        return "不存在工具：{}".format(name)
    try:
        result = fn(**(arguments if isinstance(arguments, dict) else {}))
        return result if isinstance(result, str) else str(result)
    except Exception as e:
        return "工具执行失败：{}".format(e)


def post_prompt():
    if not ALL_TOOLS:
        return "\n当前没有可用工具。"
    return (
        "\nPOST 模式工具规则：需要执行操作时只能输出一个 JSON，禁止解释，格式：\n"
        "{\"tool\":\"工具名\",\"arguments\":{}}\n"
        "禁止声称操作完成，除非收到工具返回。可用工具：{}\n工具定义：\n{}"
    ).format(", ".join(ALL_TOOL_MAP.keys()), json.dumps(ALL_TOOLS, ensure_ascii=False))


def post_messages():
    result = []
    for item in CHAT_MESSAGES:
        if item.get("role") not in ("system", "user", "assistant"):
            continue
        content = item.get("content", "")
        if not isinstance(content, str):
            content = json.dumps(content, ensure_ascii=False)
        if item["role"] == "system":
            content += post_prompt()
        result.append({"role": item["role"], "content": content})
    return result


def parse_tool(text):
    if not text:
        return None
    text = str(text).strip()
    if text.startswith("```"):
        lines = text.splitlines()[1:]
        if lines and lines[-1].strip() == "```":
            lines.pop()
        text = "\n".join(lines).strip()
    try:
        data = json.loads(text)
    except Exception:
        a, b = text.find("{"), text.rfind("}")
        if a < 0 or b <= a:
            return None
        try:
            data = json.loads(text[a:b + 1])
        except Exception:
            return None
    if not isinstance(data, dict) or not data.get("tool"):
        return None
    return {"tool": str(data["tool"]), "arguments": data.get("arguments", {})}


def call_post(api):
    headers = {"Content-Type": "application/json"}
    if api["api_key"]:
        headers["Authorization"] = "Bearer " + api["api_key"]
    body = {"messages": post_messages(), "temperature": api["temperature"]}
    with httpx.Client(trust_env=False, verify=api["verify_ssl"], timeout=120) as client:
        response = client.post(api["base_url"], headers=headers, json=body)
    data = response.json()
    if response.status_code >= 300 or data.get("success") is False:
        raise RuntimeError("HTTP {}：{}".format(response.status_code, data))
    result = data.get("result")
    if isinstance(result, dict):
        if result.get("response") is not None:
            return str(result["response"])
        if result.get("output") is not None:
            return str(result["output"])
    if isinstance(result, str):
        return result
    if data.get("response") is not None:
        return str(data["response"])
    raise RuntimeError("响应中没有找到文本：{}".format(data))


def call_opai(api):
    client = OpenAI(
        api_key=api["api_key"],
        base_url=api["base_url"],
        http_client=httpx.Client(trust_env=False, verify=api["verify_ssl"])
    )
    request = {
        "model": api["model_name"],
        "messages": CHAT_MESSAGES,
        "temperature": api["temperature"],
        "stream": True
    }
    if ALL_TOOLS:
        request["tools"] = ALL_TOOLS
        request["tool_choice"] = "auto"
    stream = client.chat.completions.create(**request)
    parts, calls = [], {}
    for chunk in stream:
        if not chunk.choices:
            continue
        delta = chunk.choices[0].delta
        if delta.content:
            print(delta.content, end="", flush=True)
            parts.append(delta.content)
        if delta.tool_calls:
            for item in delta.tool_calls:
                i = item.index
                calls.setdefault(i, {"id": "", "type": "function", "function": {"name": "", "arguments": ""}})
                if item.id:
                    calls[i]["id"] += item.id
                if item.type:
                    calls[i]["type"] = item.type
                if item.function:
                    calls[i]["function"]["name"] += item.function.name or ""
                    calls[i]["function"]["arguments"] += item.function.arguments or ""
    return "".join(parts), calls


def run_agent(query):
    global CHAT_MESSAGES
    update_time()
    api = current_api()
    CHAT_MESSAGES.append({"role": "user", "content": query})
    for _ in range(api["max_steps"]):
        try:
            if api["mode"] == "POST":
                answer = call_post(api)
                tool = parse_tool(answer)
                if tool:
                    print("\n调用工具：{}".format(tool["tool"]))
                    result = execute_tool(tool["tool"], tool["arguments"])
                    print("工具返回：{}".format(result))
                    CHAT_MESSAGES.append({"role": "assistant", "content": answer})
                    CHAT_MESSAGES.append({"role": "user", "content": "工具返回：\n{}\n请继续回答。".format(result)})
                    continue
                if any(x in answer for x in ("创建成功", "执行成功", "已删除", "已写入")):
                    return "模型声称执行了操作，但没有返回合法工具调用，实际未执行。"
                print(answer, end="", flush=True)
                CHAT_MESSAGES.append({"role": "assistant", "content": answer})
                return None
            answer, calls = call_opai(api)
            if not calls:
                CHAT_MESSAGES.append({"role": "assistant", "content": answer})
                return None
            assistant = {"role": "assistant", "content": answer, "tool_calls": list(calls.values())}
            CHAT_MESSAGES.append(assistant)
            for call in calls.values():
                result = execute_tool(call["function"]["name"], json.loads(call["function"]["arguments"] or "{}"))
                print("\n调用工具：{}\n工具返回：{}".format(call["function"]["name"], result))
                CHAT_MESSAGES.append({"role": "tool", "tool_call_id": call["id"], "content": result})
        except Exception as e:
            return "API 调用失败：{}".format(e)
    return "达到最大思考轮数。"


def clear_chat():
    global CHAT_MESSAGES
    CHAT_MESSAGES = [{"role": "system", "content": SYSTEM_PROMPT}]


def show_status():
    api = current_api()
    print("当前 API：{}，模式：{}，模型：{}，北京时间：{}".format(CURRENT_API_NAME, api["mode"], api["model_name"], now_beijing()))
    print("已加载工具：{}".format(list(ALL_TOOL_MAP.keys())))


def show_apis():
    for item in CONFIG["apis"]:
        mark = "当前" if item["name"] == CURRENT_API_NAME else ""
        print("{} API {}：{} / {}".format(mark, item["name"], item["mode"], item["model_name"]))


def main():
    global CURRENT_API_NAME
    print(LOGO)
    print("Agent：{} | 时间：{}".format(AGENT_NAME, now_beijing()))
    while True:
        try:
            text = input("<Agent-M:{}>\nUser#>".format(current_api()["model_name"])).strip()
        except (KeyboardInterrupt, EOFError):
            break
        if not text:
            continue
        cmd = text.lower()
        if cmd in ("/quit", "/stop", "stop"):
            break
        if cmd in ("/clear", "clear"):
            clear_chat()
            print("聊天记录已清空。")
            continue
        if cmd in ("/status", "status"):
            show_status()
            continue
        if cmd == "/api":
            show_apis()
            continue
        if cmd.startswith("/api "):
            name = text.split(None, 1)[1]
            if name in API_MAP:
                CURRENT_API_NAME = name
                print("已切换到 API {}。".format(name))
            else:
                print("不存在 API：{}".format(name))
            continue
        print("\nAgent：", end="")
        result = run_agent(text)
        if result:
            print("\n" + result)
        print("\n")


if __name__ == "__main__":
    main()
